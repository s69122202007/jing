📊 Graph Theory ของเส้นทาง

Vertex = จุดจอดรถเมล์ และ Edge = เส้นทางที่เชื่อมระหว่างจุดจอด

→
เยาวราช
→
สะพานพุทธ

ตัวอย่าง Adjacency List

const graph = {
  "อนุสาวรีย์": ["เยาวราช", "รังสิต"],
  "เยาวราช": ["อนุสาวรีย์", "สะพานพุทธ"],
  "สะพานพุทธ": ["เยาวราช"],
  "รังสิต": ["อนุสาวรีย์", "ดอนเมือง"],
  "ดอนเมือง": ["รังสิต"]
};

Path ตัวอย่าง: อนุสาวรีย์ → เยาวราช → สะพานพุทธ
